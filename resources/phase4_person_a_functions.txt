   ```c
   typedef struct s_command
   {
       char                **args;           // Command and arguments
       t_token             *redirects;       // List of redirect tokens (operator + filename pairs)
       struct s_command    *next;            // Next command in pipeline

   ```
   
   IMPORTANT: The redirects list contains PAIRS of tokens:
   - First token: The redirect operator (TOKEN_REDIRECT_IN, TOKEN_REDIRECT_OUT, etc.)
   - Second token: The filename (TOKEN_WORD with the file path)
   - Example: For "cat < input.txt > output.txt"
     redirects → [<]→[input.txt]→[>]→[output.txt]→NULL

   ```c
   while (tokens)
   {
       if (is_operator_token(tokens->type) && tokens->type != TOKEN_PIPE)
       {

           if (tokens->next)
               add_token_to_list(&redirects, new_token(tokens->next->type, tokens->next->value, tokens->next->quote_type));
           tokens = tokens->next;  // skip filename
       }
       tokens = tokens->next;
   }
   ```
