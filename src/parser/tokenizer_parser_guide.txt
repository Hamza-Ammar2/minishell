typedef enum e_token_type
{
    TOKEN_WORD,          // Regular word/command/argument
    TOKEN_PIPE,          // |

typedef struct s_token
{

}   t_token;

2.1 Implement new_token()
-------------------------
Purpose: Create a new token node

Pseudocode:
- Allocate memory for t_token
- If allocation fils, return NULL
- Set token->type = type
- If value is not NULL:
    * Duplicate the string (use strdup or ft_strdup)
    * If strdup fails, free token and return NULL
- Set token->value = duplicated string
- Set token->next = NULL
- Return token

Example:
t_token *token = malloc(sizeof(t_token));
token->type = TOKEN_WORD;
token->value = strdup("echo");
token->next = NULL;


2.2 Implement free_tokens()
---------------------------
Purpose: Free entire token linked list

Pseudocode:
- Create a temp pointer
- While tokens is not NULL:
    * Save tokens->next in temp
    * Free tokens->value (if not NULL)
    * Free tokens
    * Set tokens = temp

Example loop:
while (tokens)
{
    temp = tokens->next;
    free(tokens->value);
    free(tokens);
    tokens = temp;
}


2.3 Helper: add_token_to_list()
-------------------------------
Purpose: Add token to end of list

Pseudocode:
- If list is empty (*head == NULL):
    * Set *head = new_token
- Else:
    * Find last token in list
    * Set last->next = new_token

3.1 Helper: is_operator()
-------------------------
Purpose: Check if character is an operator

char *is_operator(char *str)
{
    if (*str == '|')
        return ("|");
    if (*str == '<' && *(str + 1) == '<')
        return ("<<");
    if (*str == '<')
        return ("<");
    if (*str == '>' && *(str + 1) == '>')
        return (">>");
    if (*str == '>')
        return (">");
    return (NULL);
}


3.2 Helper: get_token_type()
----------------------------
Purpose: Determine token type from operator string

t_token_type get_token_type(char *op)
{
    if (strcmp(op, "|") == 0)
        return (TOKEN_PIPE);
    if (strcmp(op, "<") == 0)
        return (TOKEN_REDIRECT_IN);
    if (strcmp(op, ">") == 0)
        return (TOKEN_REDIRECT_OUT);
    if (strcmp(op, ">>") == 0)
        return (TOKEN_REDIRECT_APPEND);
    if (strcmp(op, "<<") == 0)
        return (TOKEN_REDIRECT_HEREDOC);
    return (TOKEN_WORD);
}


3.3 Helper: extract_word()
--------------------------
Purpose: Extract a word from input until space or operator

Pseudocode:
- Start from current position
- Count characters until:
    * Space
    * Tab
    * Operator character (|, <, >)
    * End of string
- Allocate string of that length
- Copy characters into new string
- Return new string

Example:
int len = 0;
while (input[len] && !isspace(input[len]) && !is_operator(&input[len]))
    len++;
char *word = strndup(input, len);  // or implement your own


3.4 Main Function: tokenize()
-----------------------------
Purpose: Convert input string into token list

Pseudocode:
t_token *tokenize(char *input)
{
    t_token *head = NULL;
    int i = 0;
    
    while (input[i])
    {
        // Skip whitespace
        while (input[i] && isspace(input[i]))
            i++;
        
        if (!input[i])
            break;
        
        // Check for operator
        char *op = is_operator(&input[i]);
        if (op)
        {
            // Create operator token
            t_token_type type = get_token_type(op);
            t_token *token = new_token(type, NULL);
            add_token_to_list(&head, token);
            i += strlen(op);
            continue;
        }
        
        // Extract word
        char *word = extract_word(&input[i]);
        if (word)
        {
            t_token *token = new_token(TOKEN_WORD, word);
            add_token_to_list(&head, token);
            i += strlen(word);
            free(word);  // new_token duplicates it
        }
    }
    
    return (head);
}

5.3 Helper: count_args()

int count_args(t_token *tokens)
{
    int count = 0;
    while (tokens && tokens->type == TOKEN_WORD)
    {
        count++;
        tokens = tokens->next;
    }
    return (count);
}

6.2 Helper: handle_redirection()
--------------------------------
Purpose: Process redirection tokens

Pseudocode:
void handle_redirection(t_command *cmd, t_token *token)
{
    // token is the redirection operator
    // token->next should be the filename
    
    if (!token->next || token->next->type != TOKEN_WORD)
    {
        // Syntax error - redirection without file
        return (error);
    }
    
    if (token->type == TOKEN_REDIRECT_IN)
        cmd->input_file = strdup(token->next->value);
    else if (token->type == TOKEN_REDIRECT_OUT)
    {
        cmd->output_file = strdup(token->next->value);
        cmd->append_mode = 0;
    }
    else if (token->type == TOKEN_REDIRECT_APPEND)
    {
        cmd->output_file = strdup(token->next->value);
        cmd->append_mode = 1;
    }
    else if (token->type == TOKEN_REDIRECT_HEREDOC)
        cmd->heredoc_delim = strdup(token->next->value);
}


6.3 Main Function: parse() - Simple Version
-------------------------------------------
Purpose: Convert tokens into command structure (no pipes yet)

Pseudocode:
t_command *parse(t_token *tokens)
{
    if (!tokens)
        return (NULL);
    
    t_command *cmd = new_command();
    t_token *current = tokens;
    int arg_index = 0;
    
    // Count total WORD tokens (for args array size)
    int arg_count = 0;
    t_token *temp = tokens;
    while (temp && temp->type != TOKEN_PIPE)
    {
        if (temp->type == TOKEN_WORD)
            arg_count++;
        temp = temp->next;
    }
    
    // Allocate args array
    cmd->args = malloc(sizeof(char *) * (arg_count + 1));
    
    // Process tokens
    while (current && current->type != TOKEN_PIPE)
    {
        if (current->type == TOKEN_WORD)
        {
            cmd->args[arg_index++] = strdup(current->value);
        }
        else if (current->type >= TOKEN_REDIRECT_IN)
        {
            handle_redirection(cmd, current);
            current = current->next;  // Skip filename token
        }
        current = current->next;
    }
    
    cmd->args[arg_index] = NULL;  // NULL terminate
    return (cmd);
}

t_command *parse(t_token *tokens)
{

    t_command *current_cmd = NULL;
    t_token *token_start = tokens;
    
    while (tokens)
    {

        t_command *cmd = parse_single_command(&tokens);
        
        // Add to command list

            head = cmd;
        else
            current_cmd->next = cmd;

{
    char *input;
    
    while (1)
    {
        input = readline(PROMPT);

            printf("exit\n");
            break;
        }
        if (*input)
        {
            add_history(input);

                continue;

                free(input);
                continue;
            }

            // Cleanup
            free_commands(commands);
        }
        free(input);
    }

================================================================================
                            COMMON PITFALLS
================================================================================

1. Memory Leaks
   - Always free tokens after parsing
   - Always free commands after execution
   - Free all strings when freeing structures

2. NULL Checks
   - Check malloc returns
   - Check for NULL tokens
   - Check for NULL values before strdup

3. Off-by-One Errors
   - Remember NULL terminator in args array
   - Don't forget to increment i in loops

4. Operator Length
   - >> and << are 2 characters
   - Single operators are 1 character
   - Don't forget to advance i correctly

5. Redirection Handling
   - Check that filename token exists after operator
   - Handle multiple redirections (last one wins)
   - Don't forget to skip the filename token
